# Linux クライアント（vostro 等）へのリリース定義

# kisok
vostro:@home@indou@start-camera-monitor.sh /home/indou/start-camera-monitor.sh

# メトリクス収集スクリプト

vostro:@etc@systemd@system@push-metrics.service /etc/systemd/system/push-metrics.service
vostro:@etc@systemd@system@push-metrics.timer   /etc/systemd/system/push-metrics.timer
